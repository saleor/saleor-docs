Represents a type of product. It defines what attributes are available to products of this type.



```graphql
type ProductType implements Node & ObjectWithMetadata {
  id: ID!
  privateMetadata: [MetadataItem!]!
  privateMetafield(key: String!): String
  privateMetafields(keys: [String!]): Metadata
  metadata: [MetadataItem!]!
  metafield(key: String!): String
  metafields(keys: [String!]): Metadata
  name: String!
  slug: String!
  hasVariants: Boolean! @deprecated(reason: "This is a leftover from the past Simple/Configurable product distinction. Products can have multiple variants regardless of this setting. ")
  isShippingRequired: Boolean!
  isDigital: Boolean! @deprecated(reason: "Will be removed in v3.24.0, use metadata or attributes instead.")
  weight: Weight
  kind: ProductTypeKindEnum!
  products(channel: String, before: String, after: String, first: Int, last: Int): ProductCountableConnection @deprecated(reason: "Use the top-level `products` query with the `productTypes` filter.")
  taxType: TaxType @deprecated(reason: "Use `taxClass` field instead.")
  taxClass: TaxClass
  variantAttributes(variantSelection: VariantAttributeScope): [Attribute!] @deprecated(reason: "Use `assignedVariantAttributes` instead.")
  assignedVariantAttributes(variantSelection: VariantAttributeScope): [AssignedVariantAttribute!]
  productAttributes: [Attribute!]
  availableAttributes(filter: AttributeFilterInput @deprecated(reason: "Use `where` filter instead."), where: AttributeWhereInput, search: String, before: String, after: String, first: Int, last: Int): AttributeCountableConnection
}
```
