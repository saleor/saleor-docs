```graphql
type GiftCardEvent implements Node {
  id: ID!
  date: DateTime
  type: GiftCardEventsEnum
  user: User
  app: App
  message: String
  email: String
  orderId: ID
  orderNumber: String
  tags: [String!]
  oldTags: [String!]
  balance: GiftCardEventBalance
  assignedTo: GiftCardEventAssignment

  oldExpiryDate: Date
}
```
