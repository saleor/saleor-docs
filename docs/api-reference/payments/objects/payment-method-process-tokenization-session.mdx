```graphql
type PaymentMethodProcessTokenizationSession implements Event {
  issuedAt: DateTime
  version: String
  issuingPrincipal: IssuingPrincipal
  recipient: App
  user: User!
  channel: Channel!
  data: JSON
  id: String!
}
```
